        json.dumps(content,ensure_ascii=True,separators=(',',':'),sort_keys=True)+'\n'

        raise SystemExit(f'SELECTION_SNAPSHOT_NOT_CANONICAL:{path}')
    if content!=expected:
        raise SystemExit(f'SELECTION_SNAPSHOT_UNEXPECTED:{path}')
    return {
        'path':str(path),
        'sha256':hashlib.sha256(canonical).hexdigest(),
        'content':content,
    }
