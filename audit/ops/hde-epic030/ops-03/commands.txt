T4_generate_and_validate_path_proofs
python - <<'PY'
import datetime
import hashlib
from pathlib import Path
targets = [
    'audit/EPIC-030_close_report.md',
    'audit/EPIC-030_MANIFEST.json',
    'audit/ops/hde-epic030/ops-03/final_evidence_inventory.md',
]
now = datetime.datetime.utcnow().replace(microsecond=0).isoformat() + 'Z'
for raw in targets:
    p = Path(raw)
    Path(raw + '.path_proof.txt').write_text(
        'path: ' + raw + '\n'
        + 'size_bytes: ' + str(p.stat().st_size) + '\n'
        + 'sha256: ' + hashlib.sha256(p.read_bytes()).hexdigest() + '\n'
        + 'mtime_utc: ' + datetime.datetime.utcfromtimestamp(p.stat().st_mtime).replace(microsecond=0).isoformat() + 'Z\n'
        + 'produced_at_utc: ' + now + '\n',
        encoding='utf-8',
    )
for raw in targets:
    p = Path(raw)
    kv = {}
    for line in Path(raw + '.path_proof.txt').read_text(encoding='utf-8').splitlines():
        if ': ' in line:
            k, v = line.split(': ', 1)
            kv[k] = v
    if kv.get('path') != raw or kv.get('size_bytes') != str(p.stat().st_size) or kv.get('sha256') != hashlib.sha256(p.read_bytes()).hexdigest():
        raise SystemExit('path proof mismatch: ' + raw)
print('path proof generation PASS')
print('path proof validation PASS')
